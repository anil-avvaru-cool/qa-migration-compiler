## Executive Summary

**Current State:** Extraction pipeline has gaps causing data loss:
- Navigation URLs hidden in method bodies → not extracted
- Method parameters not linked to test data → input fields unresolved
- Page object context lost → targetId resolution incomplete
- Symbol table underutilized → missed semantic relationships

**Goal:** Design end-to-end extraction that captures **100% of semantic information** without workarounds.

**Approach:** Integrate missing analysis layers (Call Graph, Data Flow, Type Resolution) into a cohesive extraction pipeline.

class ParameterMapper:
    def resolve_parameter_value(
        self,
        method_call: ASTNode,      # searchProduct("Laptop")
        method_def: MethodDef,     # def searchProduct(String productName)
        parameter_index: int       # 0
    ) -> ParameterBinding:
        # Get argument at call site
        call_argument = method_call.arguments[parameter_index]
        call_value = extract_value(call_argument)
        
        # Get parameter definition
        param_def = method_def.parameters[parameter_index]
        
        # Classify origin
        if is_literal(call_value):
            return {"type": "constant", "value": call_value}
        elif is_variable(call_value):
            origin = data_flow.get_variable_origin(call_value)
            return {"type": "named_constant" if origin.is_constant else "data_field", ...}
        elif is_method_call(call_value):
            return {"type": "method_result", ...}

# src/analysis/call_graph.py

class CallGraphBuilder:
    def __init__(self, ast_trees: List[ASTTree]):
        self.ast_trees = ast_trees
        self.method_defs = {}  # {(class, method): ASTNode}
        self.method_calls = {} # {(class, method): [(class, method), ...]}
    
    def build(self):
        # Index all method definitions
        for tree in self.ast_trees:
            for node in tree.walk():
                if node.type == "method":
                    class_name = self._get_enclosing_class(node)
                    method_name = node.properties.get("name")
                    self.method_defs[(class_name, method_name)] = node
    
    def resolve_method_definition(self, class_name: str, method_name: str) -> Optional[ASTNode]:
        """Return the ASTNode for a method definition"""
        return self.method_defs.get((class_name, method_name))
    
    def resolve_method_body(self, class_name: str, method_name: str) -> List[ASTNode]:
        """Return all statements in method body"""
        method_def = self.resolve_method_definition(class_name, method_name)
        if method_def:
            return method_def.children  # Method body statements
        return []
    
    def find_internal_calls(
        self,
        class_name: str,
        method_name: str,
        target_method: str,
        target_qualifier: str = "driver"
    ) -> List[ASTNode]:
        """Find specific method calls inside another method"""
        body = self.resolve_method_body(class_name, method_name)
        calls = []
        for node in self._walk(body):
            if (node.properties.get("member") == target_method and
                node.properties.get("qualifier") == target_qualifier):
                calls.append(node)
        return calls

```python
# src/extraction/action_mapper.py

def map(self, ast_node: ASTNode, call_graph: CallGraphBuilder) -> List[Dict]:
    actions = []
    
    for node in self._walk(ast_node):
        member = node.properties.get("member")
        qualifier = node.properties.get("qualifier")
        
        # Level 1: Direct Selenium calls (existing)
        if self._is_semantic_action(member):
            actions.append(self._extract_direct_action(node))
        
        # Level 2: NEW - Page object method calls
        elif self._is_page_object_method(qualifier, member):
            # Follow method call to its definition
            method_def = call_graph.resolve_method_definition(qualifier, member)
            if method_def:
                # Extract actions from within the method
                internal_actions = self._extract_from_method_body(
                    method_def,
                    call_node=node,
                    call_graph=call_graph
                )
                actions.extend(internal_actions)
    
    return actions

def _extract_from_method_body(
    self,
    method_def: ASTNode,
    call_node: ASTNode,
    call_graph: CallGraphBuilder
) -> List[Dict]:
    """Recursively extract actions from method implementation"""
    actions = []
    
    for body_node in method_def.children:
        # Find Selenium calls inside method
        for stmt in self._walk(body_node):
            if self._is_semantic_action(stmt.properties.get("member")):
                action = self._extract_direct_action(stmt)
                
                # Map parameters using call site arguments
                action = self._map_parameters_from_call_site(
                    action,
                    call_node,
                    method_def
                )
                actions.append(action)
    
    return actions

def _map_parameters_from_call_site(
    self,
    action: Dict,
    call_node: ASTNode,
    method_def: ASTNode
) -> Dict:
    """Link extracted action parameters to actual values from call site"""
    
    # Get call arguments
    call_args = call_node.children  # searchProduct("Laptop")
    
    # Get method parameters
    method_params = [c for c in method_def.children if c.type == "parameter"]
    
    # Map: parameter name → call argument value
    param_mapping = {}
    for i, (call_arg, method_param) in enumerate(zip(call_args, method_params)):
        arg_value = call_arg.properties.get("value")
        param_name = method_param.properties.get("name")
        param_mapping[param_name] = arg_value
    
    # Apply mapping to action input
    if action.get("input"):
        input_field = action["input"].get("field")
        if input_field in param_mapping:
            action["input"]["value"] = param_mapping[input_field]
            action["input"]["source"] = "call_argument"
    
    return action
```

# src/extraction/step_enricher.py (enhance)

def _resolve_target_id(
    self,
    step: Dict,
    call_graph: Optional[CallGraphBuilder] = None
) -> Optional[str]:
    """Resolve targetId with multi-level fallback"""
    
    # Level 1: Already have targetId from extraction
    if step.get("targetId"):
        return step["targetId"]
    
    # Level 2: Use target_name_id → lookup in targets
    target_name_id = step.get("target_name_id")
    if target_name_id:
        # Try exact match
        if target_name_id in self.targets_by_id:
            return target_name_id
        if target_name_id in self.targets_by_name:
            return self.targets_by_name[target_name_id].get("targetId")
        
        # Try fuzzy match
        matched = self._fuzzy_match_target(target_name_id)
        if matched:
            return matched["targetId"]
    
    # Level 3: NEW - For navigation, resolve via call graph
    if step.get("action") == "navigate" and call_graph:
        # This is likely a navigation method call
        method_name = step.get("name")  # e.g., "navigateToProducts"
        method_def = call_graph.resolve_method_definition(
            step.get("page_class"),
            method_name
        )
        if method_def:
            url_calls = call_graph.find_internal_calls(
                step.get("page_class"),
                method_name,
                "get",
                "driver"
            )
            if url_calls:
                url = self._extract_url(url_calls[0])
                # Create implicit navigation target
                return self._generate_navigation_target_id(url)
    
    return None

def _fuzzy_match_target(self, name: str) -> Optional[Dict]:
    """Fuzzy matching for target names"""
    # Try various transformations
    candidates = [
        name,
        self._camel_to_snake(name).upper(),
        self._add_common_suffixes(name),
        # ... more strategies
    ]
    
    for candidate in candidates:
        if candidate in self.targets_by_id:
            return self.targets_by_id[candidate]
    
    return None
```

# src/ir/validator/completeness_validator.py

class CompletenessValidator:
    def validate(self, project_ir: ProjectIR) -> ValidationReport:
        report = ValidationReport()
        
        # Check 1: All step targetIds exist in targets
        for test in project_ir.tests:
            for step in test.steps:
                if step.action in ("type", "click", "select", "waitForVisible"):
                    if not step.targetId:
                        report.add_error(
                            f"Test {test.testId}, Step {step.stepId}: "
                            f"Missing targetId for action '{step.action}'"
                        )
                    elif not self._target_exists(step.targetId, project_ir.targets):
                        report.add_error(
                            f"Test {test.testId}, Step {step.stepId}: "
                            f"targetId '{step.targetId}' not found in targets.json"
                        )
                
                # Check 2: Navigation steps have target URL
                if step.action == "navigate":
                    if not step.target or step.target.get("type") != "url":
                        report.add_error(
                            f"Test {test.testId}, Step {step.stepId}: "
                            f"Navigation step must have target.type='url'"
                        )
        
        # Check 3: All inputs are resolved
        for test in project_ir.tests:
            for step in test.steps:
                if step.input:
                    if step.input.get("source") == "data":
                        if not step.input.get("field"):
                            report.add_error(
                                f"Test {test.testId}, Step {step.stepId}: "
                                f"Data source missing field name"
                            )
        
        return report

class ValidationReport:
    def __init__(self):
        self.errors = []
        self.warnings = []
        self.info = []
    
    def add_error(self, message: str):
        self.errors.append(message)
    
    def add_warning(self, message: str):
        self.warnings.append(message)
    
    def add_info(self, message: str):
        self.info.append(message)
    
    def is_complete(self) -> bool:
        return len(self.errors) == 0
    
    def to_dict(self) -> Dict:
        return {
            "complete": self.is_complete(),
            "error_count": len(self.errors),
            "warning_count": len(self.warnings),
            "errors": self.errors,
            "warnings": self.warnings,
            "info": self.info
        }
```
